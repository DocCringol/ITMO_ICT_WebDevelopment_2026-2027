    Практика 2.1. Модель, админка, первое представление

    Модель данных по рисунку 2 методички: автовладелец, автомобиль, владение и водительское удостоверение. Таблицы и поля названы латиницей, подписи для админки - через verbose_name.

    Сущность на схеме Модель Особенности
    Автовладелец CarOwner после практики 2.3 - пользователь (AbstractUser), фамилия и имя - last_name, first_name
    Автомобиль Car гос. номер, марка, модель, цвет (может быть пустым)
    Владение Ownership ассоциативная сущность: FK на владельца и машину, дата начала, дата окончания (пустая - владеет до сих пор)
    Водительское удостоверение DriverLicense FK на владельца, номер, тип (поле выбора choices: A, B, C, D), дата выдачи

    Даты на схеме помечены как datetime, но время здесь не нужно, поэтому использован DateField.

    models.py 
    from django.conf import settings
from django.contrib.auth.models import AbstractUser
from django.db import models


class CarOwner(AbstractUser):
    # Фамилия и имя берутся из AbstractUser (last_name, first_name)
    birth_date = models.DateField('Дата рождения', null=True, blank=True)
    passport_number = models.CharField('Номер паспорта', max_length=20, blank=True)
    home_address = models.CharField('Домашний адрес', max_length=200, blank=True)
    nationality = models.CharField('Национальность', max_length=50, blank=True)

    class Meta:
        verbose_name = 'Автовладелец'
        verbose_name_plural = 'Автовладельцы'

    def __str__(self):
        return f'{self.last_name} {self.first_name}'.strip() or self.username


class Car(models.Model):
    license_plate = models.CharField('Гос. номер', max_length=15)
    brand = models.CharField('Марка', max_length=20)
    model = models.CharField('Модель', max_length=20)
    color = models.CharField('Цвет', max_length=30, null=True, blank=True)
    owners = models.ManyToManyField(settings.AUTH_USER_MODEL, through='Ownership', related_name='cars')

    class Meta:
        verbose_name = 'Автомобиль'
        verbose_name_plural = 'Автомобили'

    def __str__(self):
        return f'{self.brand} {self.model} ({self.license_plate})'


class Ownership(models.Model):
    # Ассоциативная сущность: у владения есть свои атрибуты - даты начала и окончания
    owner = models.ForeignKey(settings.AUTH_USER_MODEL, on_delete=models.CASCADE, null=True,
                              related_name='ownerships', verbose_name='Владелец')
    car = models.ForeignKey(Car, on_delete=models.CASCADE, null=True,
                            related_name='ownerships', verbose_name='Автомобиль')
    start_date = models.DateField('Дата начала')
    end_date = models.DateField('Дата окончания', null=True, blank=True)

    class Meta:
        verbose_name = 'Владение'
        verbose_name_plural = 'Владения'

    def __str__(self):
        return f'{self.owner} - {self.car}'


class DriverLicense(models.Model):
    TYPES = (
        ('A', 'A - мотоциклы'),
        ('B', 'B - легковые'),
        ('C', 'C - грузовые'),
        ('D', 'D - автобусы'),
    )
    owner = models.ForeignKey(settings.AUTH_USER_MODEL, on_delete=models.CASCADE,
                              related_name='licenses', verbose_name='Владелец')
    number = models.CharField('Номер удостоверения', max_length=10)
    type = models.CharField('Тип', max_length=10, choices=TYPES)
    issue_date = models.DateField('Дата выдачи')

    class Meta:
        verbose_name = 'Водительское удостоверение'
        verbose_name_plural = 'Водительские удостоверения'

    def __str__(self):
        return f'{self.number} ({self.owner})'

    Практика 2.2. Связь М:М, представления и формы

    Связь автомобиля с владельцем - ManyToManyField с through='Ownership': у владения есть свои атрибуты (даты), поэтому промежуточная модель описана явно.

    owners = models.ManyToManyField(settings.AUTH_USER_MODEL, through='Ownership', related_name='cars')

    Представления.

    Что Тип URL
    Все владельцы функция owner_list (FBV) /owners/
    Создание владельца через форму функция owner_create (FBV) + OwnerForm /owner/create/
    Все автомобили CarListView (ListView) /cars/
    Автомобиль по id CarDetailView (DetailView) /car/<id>/
    Создание автомобиля CarCreateView (CreateView) + CarForm /car/create/
    Обновление автомобиля CarUpdateView (UpdateView) + CarForm /car/<id>/update/
    Удаление автомобиля CarDeleteView (DeleteView) /car/<id>/delete/
    views.py 
    from django.http import Http404
from django.shortcuts import redirect, render
from django.urls import reverse_lazy
from django.views.generic import DetailView, ListView
from django.views.generic.edit import CreateView, DeleteView, UpdateView

from .forms import CarForm, OwnerForm
from .models import Car, CarOwner


def owner_detail(request, owner_id):
    try:
        owner = CarOwner.objects.get(pk=owner_id)
    except CarOwner.DoesNotExist:
        raise Http404('Владелец не найден')
    return render(request, 'owner.html', {'owner': owner})


def owner_list(request):
    # Суперпользователь тоже CarOwner, но владельцем машин он не является
    owners = CarOwner.objects.filter(is_staff=False).order_by('last_name', 'first_name')
    return render(request, 'owner_list.html', {'owners': owners})


def owner_create(request):
    if request.method == 'POST':
        form = OwnerForm(request.POST)
        if form.is_valid():
            owner = form.save()
            return redirect('owner_detail', owner_id=owner.pk)
    else:
        form = OwnerForm()
    return render(request, 'owner_form.html', {'form': form})


class CarListView(ListView):
    model = Car
    template_name = 'car_list.html'


class CarDetailView(DetailView):
    model = Car
    template_name = 'car_detail.html'


class CarCreateView(CreateView):
    model = Car
    form_class = CarForm
    template_name = 'car_form.html'
    success_url = reverse_lazy('car_list')


class CarUpdateView(UpdateView):
    model = Car
    form_class = CarForm
    template_name = 'car_form.html'
    success_url = reverse_lazy('car_list')


class CarDeleteView(DeleteView):
    model = Car
    template_name = 'car_confirm_delete.html'
    success_url = reverse_lazy('car_list')

    Практика 2.3. Владелец автомобиля - пользователь

    Модель пользователя расширена через AbstractUser, как рекомендует методичка: CarOwner наследует все поля стандартного пользователя и добавляет номер паспорта, домашний адрес, национальность и дату рождения. AUTH_USER_MODEL = 'project_first_app.CarOwner' задан до первой миграции, а остальные модели ссылаются на пользователя через settings.AUTH_USER_MODEL.

    Админка. CarOwnerAdmin наследует UserAdmin и добавляет блок «Данные автовладельца» на страницы создания и редактирования.

    project_first_app/admin.py
    from django.contrib import admin
from django.contrib.auth.admin import UserAdmin

from .models import Car, CarOwner, DriverLicense, Ownership

NEW_FIELDS = ('birth_date', 'passport_number', 'home_address', 'nationality')


@admin.register(CarOwner)
class CarOwnerAdmin(UserAdmin):
    # Новые поля пользователя - отдельным блоком на странице редактирования и создания
    fieldsets = UserAdmin.fieldsets + (('Данные автовладельца', {'fields': NEW_FIELDS}),)
    add_fieldsets = UserAdmin.add_fieldsets + (
        ('Данные автовладельца', {'fields': ('first_name', 'last_name') + NEW_FIELDS}),
    )
    list_display = ('username', 'last_name', 'first_name', 'passport_number', 'nationality')


admin.site.register(Car)
admin.site.register(Ownership)
admin.site.register(DriverLicense)

    Вывод информации о пользователях. Страницы владельца и списка владельцев показывают паспорт, адрес, национальность и машины. Суперпользователь в списке не показывается: он тоже CarOwner, но машин у него нет.

    Интерфейс создания пользователя - OwnerForm наследует UserCreationForm: логин и пароль с проверкой берутся из стандартной формы, к ним добавлены новые поля. После сохранения пользователь попадает на страницу созданного владельца.
