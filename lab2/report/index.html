    Модель Назначение Связи
    User пользователь сайта (автор доклада, комментатор), расширен через AbstractUser: организация и учёная степень -
    Topic тематика М:М с Conference
    Place место проведения: название, город, адрес, описание места 1:М с Conference (PROTECT)
    Conference название, тематики, место, период (start_date, end_date), описание, условия участия -
    Registration регистрация автора на выступление: тема доклада, аннотация, секция, результат FK на User, Conference, Topic
    Review отзыв: даты конференции, текст, рейтинг 1–10, комментатор FK на Conference, User

    Решения, которые стоит пояснить:

    • Пользователь расширен через AbstractUser, как рекомендуют в практике 2.3: организация и учёная степень нужны в таблице участников, а аутентификация стандартная. AUTH_USER_MODEL = 'conferences.User' задан до первой миграции.
    • Место проведения вынесено в отдельную модель. Вариант требует хранить «описание места проведения». В одной площадке проходят разные конференции, и описание не дублируется.
    • Результат выступления - поле result с choices: «Не рассмотрено» (по умолчанию), «Рекомендован к публикации», «Не рекомендован». Ставит его только администратор.
    • Одна регистрация автора на конференцию - UniqueConstraint(author, conference). При повторной попытке сайт открывает редактирование уже существующей заявки.
    • Даты конференции в отзыве (conference_start, conference_end) копируются из конференции в момент отзыва: так они «сохраняются», как требует вариант, и отзыв остаётся привязанным к тем датам, даже если конференцию потом перенесут.
    • Рейтинг - PositiveSmallIntegerField с MinValueValidator(1) и MaxValueValidator(10).

from django.contrib.auth.models import AbstractUser
from django.core.validators import MaxValueValidator, MinValueValidator
from django.db import models

RATING_MIN = 1
RATING_MAX = 10


class User(AbstractUser):
    # Участник конференций - автор доклада или комментатор
    organization = models.CharField('Организация', max_length=200, blank=True)
    academic_degree = models.CharField('Учёная степень', max_length=100, blank=True)

    def __str__(self):
        return self.get_full_name() or self.username


class Topic(models.Model):
    name = models.CharField('Название', max_length=100, unique=True)

    class Meta:
        verbose_name = 'Тематика'
        verbose_name_plural = 'Тематики'
        ordering = ['name']

    def __str__(self):
        return self.name


class Place(models.Model):
    name = models.CharField('Название', max_length=200)
    city = models.CharField('Город', max_length=100)
    address = models.CharField('Адрес', max_length=300)
    description = models.TextField('Описание места проведения', blank=True)

    class Meta:
        verbose_name = 'Место проведения'
        verbose_name_plural = 'Места проведения'
        ordering = ['city', 'name']

    def __str__(self):
        return f'{self.name} ({self.city})'


class Conference(models.Model):
    title = models.CharField('Название', max_length=200)
    topics = models.ManyToManyField(Topic, related_name='conferences', verbose_name='Тематики')
    place = models.ForeignKey(Place, on_delete=models.PROTECT, related_name='conferences',
                              verbose_name='Место проведения')
    start_date = models.DateField('Дата начала')
    end_date = models.DateField('Дата окончания')
    description = models.TextField('Описание')
    participation_terms = models.TextField('Условия участия')

    class Meta:
        verbose_name = 'Конференция'
        verbose_name_plural = 'Конференции'
        ordering = ['start_date']

    def __str__(self):
        return self.title


class Registration(models.Model):
    # Регистрация автора на выступление; результат ставит администратор
    RESULT_PENDING = 'pending'
    RESULT_RECOMMENDED = 'recommended'
    RESULT_REJECTED = 'rejected'
    RESULTS = (
        (RESULT_PENDING, 'Не рассмотрено'),
        (RESULT_RECOMMENDED, 'Рекомендован к публикации'),
        (RESULT_REJECTED, 'Не рекомендован'),
    )

    author = models.ForeignKey(settings.AUTH_USER_MODEL, on_delete=models.CASCADE,
                               related_name='registrations', verbose_name='Автор')
    conference = models.ForeignKey(Conference, on_delete=models.CASCADE,
                                   related_name='registrations', verbose_name='Конференция')
    topic = models.ForeignKey(Topic, on_delete=models.SET_NULL, null=True, blank=True,
                              related_name='registrations', verbose_name='Секция')
    talk_title = models.CharField('Тема доклада', max_length=300)
    abstract = models.TextField('Аннотация')
    created_at = models.DateTimeField('Дата регистрации', auto_now_add=True)
    result = models.CharField('Результат выступления', max_length=20, choices=RESULTS,
                              default=RESULT_PENDING)

    class Meta:
        verbose_name = 'Регистрация на выступление'
        verbose_name_plural = 'Регистрации на выступления'
        ordering = ['conference__start_date', 'author__last_name']
        constraints = [
            # Один автор - одна регистрация на конференцию
            models.UniqueConstraint(fields=['author', 'conference'], name='unique_author_conference'),
        ]

    def __str__(self):
        return f'{self.author}: {self.talk_title}'


class Review(models.Model):
    conference = models.ForeignKey(Conference, on_delete=models.CASCADE,
                                   related_name='reviews', verbose_name='Конференция')
    author = models.ForeignKey(settings.AUTH_USER_MODEL, on_delete=models.CASCADE,
                               related_name='reviews', verbose_name='Комментатор')
    # Даты конференции на момент отзыва: если её потом перенесут, отзыв останется о тех датах
    conference_start = models.DateField('Начало конференции')
    conference_end = models.DateField('Окончание конференции')
    text = models.TextField('Текст отзыва')
    rating = models.PositiveSmallIntegerField(
        'Рейтинг', validators=[MinValueValidator(RATING_MIN), MaxValueValidator(RATING_MAX)])
    created_at = models.DateTimeField('Дата отзыва', auto_now_add=True)

    class Meta:
        verbose_name = 'Отзыв'
        verbose_name_plural = 'Отзывы'
        ordering = ['-created_at']

    def __str__(self):
        return f'{self.author} о «{self.conference}»: {self.rating}/{RATING_MAX}'

    Ключевые представления

    Список конференций: поиск, фильтрация, пагинация

    ListView с paginate_by = 4. Сколько конференций показывать на странице, пользователь выбирает сам (по 2, 4 или 8, параметр per_page): get_paginate_by берёт значение из адреса, если оно из списка, иначе остаётся 4. Фильтры берутся из GET-параметров и накладываются на queryset до пагинации, поэтому Paginator делит на страницы уже отфильтрованный список. annotate(participants=Count('registrations')) считает докладчиков одним SQL-запросом, без отдельного запроса на каждую карточку.

    conferences/views.py
    class ConferenceListView(ListView):
    model = Conference
    template_name = 'conferences/conference_list.html'
    paginate_by = 4
    # Варианты для выбора «сколько на странице»; другое значение в адресе даёт paginate_by
    per_page_choices = (2, 4, 8)

    # Сколько конференций на странице: ListView вызывает его перед тем, как делить на страницы
    def get_paginate_by(self, queryset):
        per_page = self.request.GET.get('per_page', '')
        if per_page.isdigit() and int(per_page) in self.per_page_choices:
            return int(per_page)
        return self.paginate_by

    # Какие конференции показать: здесь только строится запрос с поиском и фильтрами из адреса
    # (?q=, ?topic= ...). QuerySet ленивый, ListView дописывает к нему LIMIT/OFFSET текущей
    # страницы, и из базы приходят только её конференции - в шаблон как page_obj
    def get_queryset(self):
        queryset = (Conference.objects.select_related('place').prefetch_related('topics')
                    .annotate(participants=Count('registrations'))
                    # С annotate Meta.ordering не применяется, без порядка пагинация нестабильна
                    .order_by('start_date'))
        params = self.request.GET

        # Поиск по названию и описанию
        q = params.get('q', '').strip()
        if q:
            queryset = queryset.filter(Q(title__icontains=q) | Q(description__icontains=q))
        # Фильтры: тематика, город, прошедшие или предстоящие
        # Нечисловой id (набран руками в адресе) не фильтрует, иначе ORM падает с ValueError
        if params.get('topic', '').isdigit():
            queryset = queryset.filter(topics__id=params['topic'])
        if params.get('city'):
            queryset = queryset.filter(place__city=params['city'])
        if params.get('period') == 'upcoming':
            queryset = queryset.filter(end_date__gte=date.today())
        elif params.get('period') == 'past':
            queryset = queryset.filter(end_date__lt=date.today())
        return queryset

    Чтобы при переходе на другую страницу не терялись фильтры, ссылки пагинации строятся тегом {% querystring %} (появился в Django 5.1): он берёт текущие GET-параметры (в том числе per_page) и заменяет только page.

        <a class="page-link" href="{% querystring page=number %}">{{ number }}</a>
    </li>
{% endfor %}

    conferences/views.py
    @login_required
@require_POST
def add_review(request, pk):
    conference = get_object_or_404(Conference, pk=pk)
    form = ReviewForm(request.POST)
    if form.is_valid():
        review = form.save(commit=False)

    Таблица участников по конференциям

    Для каждой конференции нужен список её регистраций, причём с учётом фильтров (результат, поиск по ФИО, организации или теме). Prefetch с отфильтрованным queryset загружает все регистрации одним дополнительным запросом и кладёт их в атрибут filtered_registrations каждой конференции.

    conferences/views.py
    class ParticipantsView(ListView):
    # Таблица участников, сгруппированная по конференциям
    template_name = 'conferences/participants.html'
    context_object_name = 'conferences'

    # Конференции, у каждой в filtered_registrations - её заявки с учётом фильтров
    # (результат, поиск по автору и теме). Prefetch грузит заявки всех конференций одним запросом
    def get_queryset(self):
        registrations = Registration.objects.select_related('author', 'topic')
        result = self.request.GET.get('result')
        if result:
            registrations = registrations.filter(result=result)
        q = self.request.GET.get('q', '').strip()
        if q:
            registrations = registrations.filter(
                Q(author__last_name__icontains=q) | Q(author__first_name__icontains=q)

# Регистрация на сайте: GET - пустая форма, POST - создать пользователя, сразу войти
# и уйти на список конференций. При ошибках форма показывается снова с текстами ошибок
def signup(request):
    if request.method == 'POST':
        form = SignUpForm(request.POST)
        if form.is_valid():
            user = form.save()
            login(request, user)
            messages.success(request, 'Регистрация прошла успешно')
            return redirect('conference_list')
    else:
        form = SignUpForm()
    return render(request, 'registration/signup.html', {'form': form})


# Главная: список конференций с поиском, фильтрами и страницами
class ConferenceListView(ListView):
    model = Conference
    template_name = 'conferences/conference_list.html'
    paginate_by = 4
    # Варианты для выбора «сколько на странице»; другое значение в адресе даёт paginate_by
    per_page_choices = (2, 4, 8)

    # Сколько конференций на странице: ListView вызывает его перед тем, как делить на страницы
    def get_paginate_by(self, queryset):
        per_page = self.request.GET.get('per_page', '')
        if per_page.isdigit() and int(per_page) in self.per_page_choices:
            return int(per_page)
        return self.paginate_by

    # Какие конференции показать: здесь только строится запрос с поиском и фильтрами из адреса
    # (?q=, ?topic= ...). QuerySet ленивый, ListView дописывает к нему LIMIT/OFFSET текущей
    # страницы, и из базы приходят только её конференции - в шаблон как page_obj
    def get_queryset(self):
        queryset = (Conference.objects.select_related('place').prefetch_related('topics')
                    .annotate(participants=Count('registrations'))
                    # С annotate Meta.ordering не применяется, без порядка пагинация нестабильна
                    .order_by('start_date'))
        params = self.request.GET

        # Поиск по названию и описанию
        q = params.get('q', '').strip()
        if q:
            queryset = queryset.filter(Q(title__icontains=q) | Q(description__icontains=q))
        # Фильтры: тематика, город, прошедшие или предстоящие
        # Нечисловой id (набран руками в адресе) не фильтрует, иначе ORM падает с ValueError
        if params.get('topic', '').isdigit():
            queryset = queryset.filter(topics__id=params['topic'])
        if params.get('city'):
            queryset = queryset.filter(place__city=params['city'])
        if params.get('period') == 'upcoming':
            queryset = queryset.filter(end_date__gte=date.today())
        elif params.get('period') == 'past':
            queryset = queryset.filter(end_date__lt=date.today())
        return queryset

    # Что ещё передать в шаблон, кроме страницы конференций: варианты для выпадающих списков
    def get_context_data(self, **kwargs):
        context = super().get_context_data(**kwargs)
        context['topics'] = Topic.objects.all()
        context['cities'] = Place.objects.values_list('city', flat=True).distinct().order_by('city')
        context['per_page_choices'] = self.per_page_choices
        return context


# Страница одной конференции: DetailView сам находит её по pk из адреса, нет такой - 404
class ConferenceDetailView(DetailView):
    model = Conference
    template_name = 'conferences/conference_detail.html'
    queryset = Conference.objects.select_related('place').prefetch_related('topics')

    # Кроме самой конференции: отзывы, средний рейтинг, число докладчиков, пустая форма отзыва
    # и заявка текущего пользователя, чтобы показать «Изменить» вместо «Зарегистрироваться»
    def get_context_data(self, **kwargs):
        context = super().get_context_data(**kwargs)
        conference = self.object
        context['reviews'] = conference.reviews.select_related('author')
        context['avg_rating'] = conference.reviews.aggregate(avg=Avg('rating'))['avg']
        context['rating_max'] = RATING_MAX
        context['participants_count'] = conference.registrations.count()
        context['review_form'] = ReviewForm()
        if self.request.user.is_authenticated:
            context['my_registration'] = conference.registrations.filter(author=self.request.user).first()
        return context


# Новая заявка на выступление; LoginRequiredMixin отправляет анонима на страницу входа
class RegistrationCreateView(LoginRequiredMixin, CreateView):
    model = Registration
    form_class = RegistrationForm
    template_name = 'conferences/registration_form.html'

    # Вызывается первым на любой запрос: находим конференцию из адреса, а если заявка
    # на неё уже есть - сразу уводим на её редактирование (одна заявка на конференцию)
    def dispatch(self, request, *args, **kwargs):
        self.conference = get_object_or_404(Conference, pk=kwargs['pk'])
        if request.user.is_authenticated:
            existing = self.conference.registrations.filter(author=request.user).first()
            if existing:
                messages.info(request, 'Вы уже зарегистрированы, можно изменить заявку')
                return redirect('registration_update', pk=existing.pk)
        return super().dispatch(request, *args, **kwargs)

    # Аргументы для RegistrationForm: передаём конференцию, чтобы в списке секций были только её тематики
    def get_form_kwargs(self):
        kwargs = super().get_form_kwargs()
        kwargs['conference'] = self.conference
        return kwargs

    # Конференция нужна шаблону для заголовка «Заявка на ...»
    def get_context_data(self, **kwargs):
        context = super().get_context_data(**kwargs)
        context['conference'] = self.conference
        return context

    # Форма прошла проверку: автора и конференцию ставим сами, в форме их нет, подделать нельзя
    def form_valid(self, form):
        form.instance.author = self.request.user
        form.instance.conference = self.conference
        messages.success(self.request, 'Вы зарегистрированы на выступление')
        return super().form_valid(form)

    # Куда уйти после сохранения
    def get_success_url(self):
        return reverse('my_registrations')

# Отзыв к конференции: только вошедшим и только POST. Даты конференции копируются в отзыв,
# автор и конференция ставятся здесь. После - назад на страницу конференции
@login_required
@require_POST
def add_review(request, pk):
    conference = get_object_or_404(Conference, pk=pk)
    form = ReviewForm(request.POST)
    if form.is_valid():
        review = form.save(commit=False)
        review.conference = conference
        review.author = request.user
        review.conference_start = conference.start_date
        review.conference_end = conference.end_date
        review.save()
        messages.success(request, 'Спасибо за отзыв')
    else:
        messages.error(request, f'Отзыв не сохранён: рейтинг от {RATING_MIN} до {RATING_MAX} и непустой текст')
    return redirect('conference_detail', pk=pk)

class ParticipantsView(ListView):
    # Таблица участников, сгруппированная по конференциям
    template_name = 'conferences/participants.html'
    context_object_name = 'conferences'

    # Конференции, у каждой в filtered_registrations - её заявки с учётом фильтров
    # (результат, поиск по автору и теме). Prefetch грузит заявки всех конференций одним запросом
    def get_queryset(self):
        registrations = Registration.objects.select_related('author', 'topic')

        if q:
            registrations = registrations.filter(
                Q(author__last_name__icontains=q) | Q(author__first_name__icontains=q)
                | Q(author__organization__icontains=q) | Q(talk_title__icontains=q))

        conferences = Conference.objects.prefetch_related(
            Prefetch('registrations', queryset=registrations, to_attr='filtered_registrations'))
        if self.request.GET.get('conference', '').isdigit():
            conferences = conferences.filter(pk=self.request.GET['conference'])
        return conferences

    # Варианты для выпадающих списков фильтров: все конференции и все результаты
    def get_context_data(self, **kwargs):

    Разграничение прав администратора и пользователя

    Администратор (is_staff) видит и делает больше прямо на сайте, а не только в Django-admin. Главное здесь - последний пункт варианта: результат выступления администратор ставит не только в Django-admin (list_editable), но и в таблице участников на сайте (set_result).

    Отзыв к конференции нет да да
